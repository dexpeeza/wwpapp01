บันทึกใบงานปัจจุบันเข้าสู่ระบบ & Google Sheets

ระบบจะบันทึกข้อมูลทั้งในเครื่อง (เปิดดูและแก้ไขซ้ำได้ทันที) และส่งไปยัง Google Sheets Web App URL หากตั้งค่าไว้

วิธีเชื่อม Google Sheets ของคุณเองง่ายๆ:

  1. สร้างไฟล์ Google Sheets ใหม่ ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  2. คัดลอกโค้ดด้านล่างไปวางทับทั้งหมด แล้วกด การทำให้ใช้งานได้ (Deploy) > รายการใหม่ (New deployment)
  3. เลือกประเภทเป็น เว็บแอป (Web App) ตั้งค่าผู้มีสิทธิ์เข้าถึงเป็น ทุกคน (Anyone) แล้วนำ URL มาวางในช่องด้านบน
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Timestamp", "JobNo", "MachineCode", "MachineName", "ServiceDate", "Status", "Technician", "Temp1_C", "Temp2_C", "Temp3_C", "Vib1_mms", "Vib2_mms", "Vib3_mms", "AlarmCode", "OverallResult", "Comment", "SectionsSummary"]);
  }
  var d = JSON.parse(e.postData.contents);
  sheet.appendRow([new Date(), d.jobNo, d.machineCode, d.machineName, d.serviceDate, d.status, d.technician, d.temp1, d.temp2, d.temp3, d.vib1, d.vib2, d.vib3, d.alarmCode, d.overallResult, d.comment, d.sectionsSummary]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
